## 7. Verification

Independent check script (`w7_verify.py`, stdlib only) re-parses the backups and all outputs from
bytes. It asserts: backup identity; the 574 = 372 + 158 + 44 block split; every November block
byte-identical and in order; the archive body equals the concatenated source byte ranges (sha256);
the 44 overlap matches; no block is both live and archived; `live == original minus removed spans
plus the pointer block`; BOM/CRLF; the one non-`CONTINUITY`/`ISSUES` difference is the pointer
block; and the manifest hashes.

```python
#!/usr/bin/env python
"""W7 independent verification: byte-exact rotation check.

Reads the pre-rotation backups, the live files, the new archive and the manifest.
No imports from the rotation script; re-parses everything from bytes.
"""
import hashlib
import json
import os
import re

ROOT = r"C:\Users\hunte\Hrns\hornlab-search"
DOCS = os.path.join(ROOT, "docs")
HIST = os.path.join(DOCS, "history")
BACK = r"C:\Users\hunte\AppData\Local\Temp\opencode\w7_backup"
BOM = b"\xef\xbb\xbf"
DATE_RE = re.compile(r"^## (\d{4}-\d{2}-\d{2})")
CUTOFF = "2026-11-01"

ok = True


def check(name, cond, detail=""):
    global ok
    ok = ok and bool(cond)
    print(("PASS  " if cond else "FAIL  ") + name + (("  " + detail) if detail else ""))


def sha(b):
    return hashlib.sha256(b).hexdigest()


def load(path, strict_crlf=False):
    raw = open(path, "rb").read()
    bom = raw.startswith(BOM)
    body = raw[len(BOM):] if bom else raw
    if strict_crlf:
        lone = body.replace(b"\r\n", b"").count(b"\n")
        check("CRLF only in " + os.path.basename(path), lone == 0, "%d lone LF" % lone)
    return raw, body.decode("utf-8"), bom, body


def repair(s):
    try:
        return s.encode("cp1252").decode("utf-8")
    except (UnicodeEncodeError, UnicodeDecodeError):
        return s


def parse(lines):
    starts = [i for i, l in enumerate(lines) if DATE_RE.match(l)]
    out = []
    for n, s in enumerate(starts):
        e = starts[n + 1] if n + 1 < len(starts) else len(lines)
        for j in range(s + 1, e):
            if lines[j].startswith("## Status") or lines[j].startswith("# "):
                e = j
                break
        out.append({"start": s, "end": e, "date": DATE_RE.match(lines[s]).group(1),
                    "text": "\r\n".join(lines[s:e])})
    return out


def offsets(lines):
    o = [0]
    for l in lines:
        o.append(o[-1] + len(l) + 2)  # CRLF
    return o


# ---- inputs
orig_raw, orig_txt, orig_bom, orig_body = load(os.path.join(BACK, "CONTINUITY.md"), True)
live_raw, live_txt, live_bom, live_body = load(os.path.join(DOCS, "CONTINUITY.md"), True)
old_raw, old_txt, old_bom, _ = load(os.path.join(HIST, "CONTINUITY_2026-10-12_2026-10-22.md"))
new_raw, new_txt, new_bom, new_body = load(os.path.join(HIST, "CONTINUITY_2026-10-23_2026-10-31.md"), True)
iss_orig_raw, iss_orig_txt, iss_orig_bom, _ = load(os.path.join(BACK, "ISSUES.md"), True)
iss_raw, iss_txt, iss_bom, iss_body = load(os.path.join(DOCS, "ISSUES.md"), True)
man = json.load(open(os.path.join(HIST, "manifest.json"), encoding="utf-8"))

print("== backup identity ==")
check("CONTINUITY.md backup sha256", sha(orig_raw) ==
      "b8fd979ec337a234bd0a19e52bb53c53d9cc082501354988ce730a8656e9f1be", sha(orig_raw))
check("ISSUES.md backup sha256", sha(iss_orig_raw) ==
      "04b6f83a1fa29b670cc44dd8f9ae25d055216376133bd07b14bc67a0e2575833", sha(iss_orig_raw))

# ---- block accounting
orig_blocks = parse(orig_txt.split("\r\n"))
live_blocks = parse(live_txt.split("\r\n"))
new_blocks = parse(new_txt.split("\r\n"))
old_blocks = parse(old_txt.replace("\r\n", "\n").split("\n"))
orig_nov = [b for b in orig_blocks if b["date"] >= CUTOFF]
orig_pre = [b for b in orig_blocks if b["date"] < CUTOFF]

print("\n== CONTINUITY ==")
check("original 574 dated blocks", len(orig_blocks) == 574, str(len(orig_blocks)))
check("live keeps every November entry (372)", len(live_blocks) == len(orig_nov) == 372,
      "%d live / %d source" % (len(live_blocks), len(orig_nov)))
check("live has no pre-November dated block", all(b["date"] >= CUTOFF for b in live_blocks))
check("live November blocks byte-identical and in order",
      [b["text"] for b in live_blocks] == [b["text"] for b in orig_nov])
check("new archive covers 2026-10-23..2026-10-31, 158 blocks",
      len(new_blocks) == 158 and min(b["date"] for b in new_blocks) == "2026-10-23"
      and max(b["date"] for b in new_blocks) == "2026-10-31",
      "%d blocks" % len(new_blocks))
check("new archive distinct blocks", len({b["text"] for b in new_blocks}) == 140,
      str(len({b["text"] for b in new_blocks})))

# 44 already-archived blocks: repaired text == old archive text
old_norm = {"\n".join(b["text"].split("\r\n")).rstrip("\n") for b in old_blocks}
already = []
for b in orig_pre:
    if b["date"] > "2026-10-22":
        continue
    rep = "\n".join(repair(x) for x in b["text"].split("\r\n")).rstrip("\n")
    if rep in old_norm:
        already.append(b)
check("44 live pre-Oct-23 blocks already in existing archive (repaired match)",
      len(already) == 44 and len(old_blocks) == 44, "%d matched" % len(already))

# ---- byte-exact proof: archive body == concatenated source slices
orig_lines = orig_txt.split("\r\n")
offs = offsets(orig_lines)
raw_slices = []
for b in orig_pre:
    if b["date"] > "2026-10-22":
        raw_slices.append(orig_txt[offs[b["start"]]:offs[b["end"]]])
concat_new = "".join(raw_slices).encode("utf-8")
body_start = new_raw.index(b"## 2026-")
file_body = new_raw[body_start:]
check("removed bytes == archived bytes (sha256)",
      sha(concat_new) == sha(file_body),
      "removed %s  archived %s" % (sha(concat_new)[:16], sha(file_body)[:16]))
check("archive body has no extra/missing bytes", len(concat_new) == len(file_body),
      "%d vs %d" % (len(concat_new), len(file_body)))
check("archive header is the only added content", new_raw[:body_start].decode("utf-8").count("\r\n") == 6)

# every archived block appears in the live source exactly as removed
check("each new-archive block is byte-identical to its source slice",
      all(b["text"] in orig_txt for b in new_blocks))

# ---- no rotation-introduced duplication
live_texts = {b["text"] for b in live_blocks}
check("no new-archive block remains in the live file",
      not any(b["text"] in live_body.decode("utf-8") for b in new_blocks))
check("no already-archived block remains in the live file",
      not any(b["text"] in live_body.decode("utf-8") for b in already))

# ---- reverse patch: live minus Archive block == original minus removed spans
ARCH_START = live_txt.index("## Archive")
ARCH_END = live_txt.index("## 2026-11-25 ")  # first block after the pointer
without = live_txt[:ARCH_START] + live_txt[ARCH_END:]
drop = set()
for b in orig_pre:
    drop.update(range(b["start"], b["end"]))
expected = "\r\n".join(l for i, l in enumerate(orig_lines) if i not in drop)
check("live == original with pre-November spans removed (+ pointer block)",
      without == expected, "len %d vs %d" % (len(without), len(expected)))

print("\n== files ==")
check("CONTINUITY starts with BOM + '# CONTINUITY'",
      live_raw.startswith(BOM + b"# CONTINUITY"), repr(live_raw[:20]))
check("ISSUES starts with BOM + '# ISSUES'", iss_raw.startswith(BOM + b"# ISSUES"), repr(iss_raw[:20]))
check("new archive has no BOM, starts with '# CONTINUITY 2026-10-23'",
      not new_bom and new_txt.startswith("# CONTINUITY 2026-10-23"), repr(new_raw[:30]))
check("ISSUES unchanged except the pointer block",
      iss_txt.replace(iss_txt[iss_txt.index("## Archive"):iss_txt.index("## 2026-11-25 ")], "")
      == iss_orig_txt)

iss_blocks = parse(iss_txt.split("\r\n"))
check("ISSUES: 169 dated sections, all November",
      len(iss_blocks) == 169 and min(b["date"] for b in iss_blocks) >= CUTOFF,
      "%d sections, min %s" % (len(iss_blocks), min(b["date"] for b in iss_blocks)))

# ---- manifest
print("\n== manifest ==")
by_file = {a["file"]: a for a in man["archives"]}
check("manifest lists the new archive",
      "docs/history/CONTINUITY_2026-10-23_2026-10-31.md" in by_file)
check("manifest sha matches new archive",
      by_file["docs/history/CONTINUITY_2026-10-23_2026-10-31.md"]["sha256"] == sha(new_raw))
check("manifest sha matches old archive",
      by_file["docs/history/CONTINUITY_2026-10-12_2026-10-22.md"]["sha256"] == sha(old_raw))
check("manifest live sha matches CONTINUITY",
      man["live"]["docs/CONTINUITY.md"]["sha256"] == sha(live_raw))
check("manifest live sha matches ISSUES",
      man["live"]["docs/ISSUES.md"]["sha256"] == sha(iss_raw))

print("\n== sizes ==")
for label, raw in (("CONTINUITY live", live_raw), ("ISSUES live", iss_raw),
                   ("CONTINUITY new archive", new_raw)):
    n = raw.count(b"\r\n")
    nb = sum(1 for l in raw.split(b"\r\n") if l.strip())
    print("  %-24s %9d B  %6d lines  %6d non-blank  sha256 %s"
          % (label, len(raw), n, nb, sha(raw)[:16]))

print("\nRESULT:", "ALL CHECKS PASS" if ok else "FAILURES PRESENT")
raise SystemExit(0 if ok else 1)
```
